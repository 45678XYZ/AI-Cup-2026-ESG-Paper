% Figure 1: the label hierarchy and the alternative decision routes.
%
% Build with analysis/figure1.py (or `python -m analysis`), never by hand:
% the counts come from figure1_defs.tex, which is generated from paper.labels.
% Nothing in this file may type a count out; tests/test_analysis_figure1.py
% fails the build if it does.
%
% Fonts are the ACM template's own, so the figure's type matches the paper's.
% Panel coordinates run 0-10 and are scaled once, which keeps the geometry
% readable while leaving the type at its true size.
%
% For D: the natural size is 15.4 x 7.0 cm (6.1 x 2.8 in), meant for figure*
% across both columns. Include it without a width key -- at 1:1 the labels sit
% at 8pt against the template's 9pt body. Forcing width=\textwidth enlarges
% them past the running text, and no single-column reduction leaves them
% legible; this figure carries too much to fit one column.
\documentclass[border=1pt]{standalone}
% Pinned so two builds produce the same bytes: pdfTeX derives the trailer /ID
% from the output path and the clock. The date half is pinned by
% SOURCE_DATE_EPOCH in analysis/figure1.py.
\ifdefined\pdftrailerid\pdftrailerid{}\fi
% SOURCE_DATE_EPOCH reaches latexmk but the standalone class does not honour it,
% so the timestamp is omitted outright instead. A figure whose bytes change on
% every rebuild cannot be committed, and this one is.
\ifdefined\pdfinfoomitdate\pdfinfoomitdate=1\fi

\usepackage[T1]{fontenc}
\usepackage[tt=false,type1=true]{libertine}
\usepackage[libertine]{newtxmath}
\usepackage{tikz}
\usetikzlibrary{arrows.meta}

\input{figure1_defs}

\definecolor{ink}{HTML}{333333}
\definecolor{routegrey}{HTML}{8C8C8C}
\definecolor{routenone}{HTML}{4D4D4D}
\definecolor{routeglobal}{HTML}{1F77B4}
\definecolor{routecond}{HTML}{D62728}

\tikzset{
  box/.style     = {draw=ink, rounded corners=1.6pt, line width=0.35pt,
                    fill=white, align=center, inner sep=3.2pt, text=ink},
  % Parallel alternatives are drawn to one width. Letting TikZ size each box
  % to its own label makes "projection" narrower than its neighbours, which
  % reads as a lesser option rather than an equal one.
  choice/.style  = {box, minimum width=2.05cm},
  leaf/.style    = {box, minimum width=1.35cm},
  edgelabel/.style = {text=routegrey, inner sep=1.2pt, font=\scriptsize},
  note/.style    = {text=ink, align=center},
  aside/.style   = {text=routegrey, align=center, font=\itshape\scriptsize},
  methods/.style = {text=ink, font=\bfseries\scriptsize},
  % Panel (b) is a factorial, so it is drawn as a grid: calibration across,
  % output rule down. One box per cell, all cells the same size, so no two
  % boxes can line up in a way that suggests a pairing the design does not have.
  % Node sizes are in cm and the picture is scaled 0.70, so the column pitch
  % has to clear the box width after scaling: 2.5 units x 0.70 = 1.75cm
  % against a 1.5cm box leaves a 0.25cm gutter. Getting this wrong makes the
  % cells abut and the grid read as one bar.
  colhead/.style = {box, minimum width=1.5cm, minimum height=0.72cm},
  rowhead/.style = {text=ink, align=right, anchor=east, font=\scriptsize},
  cell/.style    = {box, minimum width=1.5cm, minimum height=0.62cm,
                    font=\bfseries},
  cellna/.style  = {draw=routegrey, densely dotted, rounded corners=1.6pt,
                    line width=0.35pt, align=center, inner sep=3.2pt,
                    text=routegrey, minimum width=1.5cm, minimum height=0.62cm},
  title/.style   = {text=ink, anchor=west, font=\small},
  route/.style   = {draw, line width=0.32pt, -{Stealth[length=2.6pt,width=2pt]}},
  alt/.style     = {route, draw=routegrey, densely dashed},
}

\begin{document}
\begin{tikzpicture}[scale=0.70, font=\footnotesize, every node/.style={inner sep=2pt}]

% ---------------------------------------------------------------- panel (a)
\node[title] at (-0.4, 10.1) {(a) Task hierarchy};

\node[box] (ps)   at (5.0, 8.9) {promise\_status\\Yes / No};
\node[box] (psno) at (1.8, 6.6) {VT = ES = EQ\\= N/A};
\node[box] (psyes) at (6.9, 6.6)
  {verification\_timeline (\NumTimelines)\\evidence\_status (Yes / No)};

\draw[route] (ps) -- (psno);
\draw[route] (ps) -- (psyes);
\node[edgelabel] at (2.9, 7.95) {No};
\node[edgelabel] at (6.4, 7.95) {Yes};

\node[leaf] (esno)  at (5.7, 4.2) {EQ = N/A};
\node[leaf] (esyes) at (8.6, 4.2) {EQ (\NumQualities)};

\draw[route] (psyes) -- (esno);
\draw[route] (psyes) -- (esyes);
\node[edgelabel] at (5.15, 5.4)  {ES = No};
\node[edgelabel] at (8.95, 5.4) {ES = Yes};

\node[note] at (5.0, 2.4)
  {$1 + \NumTimelines\times(1+\NumQualities) = \NumStates$ legal tuples};
\node[note] at (5.0, 1.6)
  {of $\NumPromiseLabels\times\NumTimelineLabels\times\NumEvidenceLabels
      \times\NumQualityLabels = \NumCombinations$ combinations};
\node[aside] at (5.0, 0.8) {the other \NumInvalid{} are invalid};

% ---------------------------------------------------------------- panel (b)
\begin{scope}[shift={(11.5, 0)}]
\node[title] at (-0.4, 10.1)
  {(b) Decision rules: output rule $\times$ calibration};

% Calibration is the column, output rule is the row. The previous drawing put
% one box per calibration in a row directly above one box per output rule, and
% the three columns lined up: "global bias" sat over "projection" and
% "conditional bias" over the state decoder, which reads as three pipelines.
% The design is a 2x3 factorial -- every calibration is available to both
% structured rules -- so it is drawn as the grid it is.
\def\colnone{3.95}
\def\colglobal{6.45}
\def\colcond{8.95}
\def\rowargmax{6.35}
\def\rowproj{4.95}
\def\rowstate{3.55}

\node[box] (base) at (\colglobal, 8.9)
  {base probabilities\\(identical for every method)};

\node[colhead, draw=routenone]   at (\colnone, 7.75)  {no\\calibration};
\node[colhead, draw=routeglobal] at (\colglobal, 7.75) {global\\bias};
\node[colhead, draw=routecond]   at (\colcond, 7.75)  {conditional\\bias};

\draw[route] (base) -- (\colglobal, 8.2);

\node[rowhead] at (2.55, \rowargmax) {independent argmax\\\emph{may be invalid}};
\node[rowhead] at (2.55, \rowproj)   {projection\\\emph{always valid}};
\node[rowhead] at (2.55, \rowstate)  {\NumStates-state decoding\\\emph{always valid}};

% M0 is the uncalibrated baseline and exists in one cell only; the other two
% cells of its row are drawn empty rather than left blank, so that the reader
% sees a deliberate gap instead of a rendering fault.
\node[cell]   at (\colnone, \rowargmax)   {M0};
\node[cellna] at (\colglobal, \rowargmax) {--};
\node[cellna] at (\colcond, \rowargmax)   {--};

\node[cell] at (\colnone, \rowproj)   {M1};
\node[cell] at (\colglobal, \rowproj) {M2};
\node[cell] at (\colcond, \rowproj)   {M3};

\node[cell] at (\colnone, \rowstate)   {M4};
\node[cell] at (\colglobal, \rowstate) {M5};
\node[cell] at (\colcond, \rowstate)   {M6};

\node[aside] at (\colglobal, 2.1)
  {the seven rules are alternatives, not stages:\\
   every cell consumes the same base probabilities};
\end{scope}

\end{tikzpicture}
\end{document}
